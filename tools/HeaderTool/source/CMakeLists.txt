cmake_minimum_required(VERSION 3.15 FATAL_ERROR)

project("VikingHeaderTool")

# Declare custom configurations for multi-config generators (e.g. Visual Studio)
set(CMAKE_CONFIGURATION_TYPES Debug Development Shipping CACHE TYPE INTERNAL FORCE)

# Make Development and Shipping inherit Release flags
set(CMAKE_C_FLAGS_DEVELOPMENT "${CMAKE_C_FLAGS_RELWITHDEBINFO}")
set(CMAKE_CXX_FLAGS_DEVELOPMENT "${CMAKE_CXX_FLAGS_RELWITHDEBINFO}")
set(CMAKE_EXE_LINKER_FLAGS_DEVELOPMENT "${CMAKE_EXE_LINKER_FLAGS_RELWITHDEBINFO}")
set(CMAKE_SHARED_LINKER_FLAGS_DEVELOPMENT "${CMAKE_SHARED_LINKER_FLAGS_RELWITHDEBINFO}")

set(CMAKE_C_FLAGS_SHIPPING "${CMAKE_C_FLAGS_RELEASE}")
set(CMAKE_CXX_FLAGS_SHIPPING "${CMAKE_CXX_FLAGS_RELEASE}")
set(CMAKE_EXE_LINKER_FLAGS_SHIPPING "${CMAKE_EXE_LINKER_FLAGS_RELEASE}")
set(CMAKE_SHARED_LINKER_FLAGS_SHIPPING "${CMAKE_SHARED_LINKER_FLAGS_RELEASE}")

# Add global preprocessor definitions for all targets
add_compile_definitions(
    $<$<CONFIG:Debug>:BUILD_CONFIG_DEBUG>
    $<$<CONFIG:Development>:BUILD_CONFIG_DEVELOPMENT>
    $<$<CONFIG:Shipping>:BUILD_CONFIG_SHIPPING>
)

# Support for single-config generators (e.g. Ninja, Makefiles)
if(NOT CMAKE_CONFIGURATION_TYPES AND CMAKE_BUILD_TYPE)
    message(STATUS "Single-config generator with build type: ${CMAKE_BUILD_TYPE}")
    if(CMAKE_BUILD_TYPE STREQUAL "Development")
        set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS_RELWITHDEBINFO}")
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS_RELWITHDEBINFO}")
        set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS_RELWITHDEBINFO}")
        set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS_RELWITHDEBINFO}")
        add_compile_definitions(BUILD_CONFIG_DEVELOPMENT)
    elseif(CMAKE_BUILD_TYPE STREQUAL "Shipping")
        set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS_RELEASE}")
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS_RELEASE}")
        set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS_RELEASE}")
        set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS_RELEASE}")
        add_compile_definitions(BUILD_CONFIG_SHIPPING)
    endif()
endif()

message(STATUS "Detected Windows SDK version: ${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}")
if (CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION VERSION_LESS "10.0.26100.0")
    message(FATAL_ERROR "Windows SDK version 10.0.26100.0 or newer is required.")
endif()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/lib")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/lib")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)

set(VKR_HEADER_TOOL_SRC_DIR "${PROJECT_SOURCE_DIR}/src")

add_subdirectory(src)
